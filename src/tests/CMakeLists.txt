function(ouaexp_configure_test_environment test_name)
    if(WIN32)
        set(test_home "${CMAKE_BINARY_DIR}/.test-home")
        set(test_appdata "${CMAKE_BINARY_DIR}/.test-appdata")
        set(test_environment
            "QT_QPA_PLATFORM=offscreen"
            "QT_FORCE_STDERR_LOGGING=1"
            "USERPROFILE=${test_home}"
            "HOME=${test_home}"
            "APPDATA=${test_home}/AppData/Roaming"
            "LOCALAPPDATA=${test_appdata}")
        set(environment_modifications
            "PATH=path_list_prepend:${QT_BINARY_DIR}"
            "PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt${QT_VERSION_MAJOR}::OpcUa>"
            "QT_PLUGIN_PATH=path_list_prepend:${QT_BINARY_DIR}/../plugins")
        if(DEFINED QTOPCUA_INSTALL_DIR)
            list(APPEND environment_modifications
                "QT_PLUGIN_PATH=path_list_prepend:${QTOPCUA_INSTALL_DIR}/plugins")
        endif()
        if(TARGET OpenSSL::Crypto)
            list(APPEND environment_modifications
                "PATH=path_list_prepend:$<TARGET_FILE_DIR:OpenSSL::Crypto>")
        endif()
        if(OPENSSL_ROOT_DIR)
            list(APPEND environment_modifications
                "PATH=path_list_prepend:${OPENSSL_ROOT_DIR}/bin")
        endif()
        set_tests_properties(${test_name} PROPERTIES
            ENVIRONMENT "${test_environment}"
            ENVIRONMENT_MODIFICATION "${environment_modifications}")
    elseif(APPLE)
        set(environment_modifications
            "QT_PLUGIN_PATH=path_list_prepend:${QT_BINARY_DIR}/../plugins")
        if(DEFINED QTOPCUA_INSTALL_DIR)
            list(APPEND environment_modifications
                "QT_PLUGIN_PATH=path_list_prepend:${QTOPCUA_INSTALL_DIR}/plugins")
        endif()
        if(TARGET OpenSSL::Crypto)
            list(APPEND environment_modifications
                "DYLD_LIBRARY_PATH=path_list_prepend:$<TARGET_FILE_DIR:OpenSSL::Crypto>")
        endif()
        set_tests_properties(${test_name} PROPERTIES
            ENVIRONMENT "QT_QPA_PLATFORM=offscreen"
            ENVIRONMENT_MODIFICATION "${environment_modifications}")
    else()
        set(environment_modifications
            "LD_LIBRARY_PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt${QT_VERSION_MAJOR}::Core>"
            "LD_LIBRARY_PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt${QT_VERSION_MAJOR}::OpcUa>"
            "QT_PLUGIN_PATH=path_list_prepend:${QT_BINARY_DIR}/../plugins")
        if(DEFINED QTOPCUA_INSTALL_DIR)
            list(APPEND environment_modifications
                "QT_PLUGIN_PATH=path_list_prepend:${QTOPCUA_INSTALL_DIR}/plugins")
        endif()
        if(TARGET OpenSSL::Crypto)
            list(APPEND environment_modifications
                "LD_LIBRARY_PATH=path_list_prepend:$<TARGET_FILE_DIR:OpenSSL::Crypto>")
        endif()
        set_tests_properties(${test_name} PROPERTIES
            ENVIRONMENT "QT_QPA_PLATFORM=offscreen"
            ENVIRONMENT_MODIFICATION "${environment_modifications}")
    endif()
endfunction()

# Everything that shapes a test compile line. The shared precompiled header is
# only reusable while the provider and the tests agree on all of it, so both go
# through this one function rather than repeating the settings.
function(ouaexp_apply_test_compile_settings target)
    # Qt Widgets is linked by every test, not just the UI ones: it is what makes
    # QTEST_MAIN instantiate a QApplication, and QT_WIDGETS_LIB has to be defined
    # identically everywhere or the shared precompiled header stops being reusable.
    target_link_libraries(${target} PRIVATE
        Qt${QT_VERSION_MAJOR}::Test
        Qt${QT_VERSION_MAJOR}::Widgets
    )
    target_compile_definitions(${target} PRIVATE
        OUAEXP_TEST_SERVER_SCRIPT="${PROJECT_SOURCE_DIR}/../tools/opcua_test_server.py")
    ouaexp_apply_coverage(${target})
endfunction()

# Defines the objects that are compiled once and handed to the test executables,
# so that ~70 targets do not each pay for the same translation unit. Must run
# before any test is registered: ouaexp_register_test() reuses the precompiled
# header of ouaexp_test_pch, and REUSE_FROM needs that target to already exist.
function(ouaexp_configure_shared_test_objects)
    # Redirects QSettings into a per-process temporary directory; every test links it.
    add_library(ouaexp_test_settings_isolation OBJECT
        "${CMAKE_CURRENT_SOURCE_DIR}/testsettingsisolation.cpp"
    )
    target_link_libraries(ouaexp_test_settings_isolation PRIVATE Qt${QT_VERSION_MAJOR}::Core)

    # The application resources are a few megabytes of generated byte arrays, which
    # costs seconds of compile time per target. The UI tests take this object instead.
    add_library(ouaexp_test_resources OBJECT "${PROJECT_SOURCE_DIR}/app/resources.qrc")
    target_link_libraries(ouaexp_test_resources PRIVATE Qt${QT_VERSION_MAJOR}::Core)

    # Provider for the precompiled header shared by all test executables. ouaexp_core
    # is what carries /utf-8 to every test, so the provider links it too.
    add_library(ouaexp_test_pch OBJECT "${CMAKE_CURRENT_SOURCE_DIR}/testpch.cpp")
    ouaexp_apply_test_compile_settings(ouaexp_test_pch)
    target_link_libraries(ouaexp_test_pch PRIVATE ouaexp_core)
    ouaexp_create_test_pch(ouaexp_test_pch)
endfunction()

# Pass NO_PCH for a test that needs compile definitions of its own; it then
# compiles without the shared precompiled header instead of failing to reuse it.
function(ouaexp_register_test test_name source_file)
    add_executable(${test_name} "${CMAKE_CURRENT_SOURCE_DIR}/${source_file}")
    target_sources(${test_name} PRIVATE $<TARGET_OBJECTS:ouaexp_test_settings_isolation>)
    ouaexp_apply_test_compile_settings(${test_name})
    if(NOT "NO_PCH" IN_LIST ARGN)
        ouaexp_reuse_test_pch(${test_name} ouaexp_test_pch)
    endif()
    ouaexp_copy_qtopcua_runtime(${test_name})
    add_test(NAME ${test_name} COMMAND ${test_name})
    set_tests_properties(${test_name} PROPERTIES LABELS "ouaexp")
    ouaexp_configure_test_environment(${test_name})
    set_property(GLOBAL APPEND PROPERTY OUAEXP_TEST_TARGETS ${test_name})
endfunction()

function(ouaexp_add_test test_name source_file)
    ouaexp_register_test(${test_name} ${source_file} ${ARGN})
    target_link_libraries(${test_name} PRIVATE ouaexp_core)
endfunction()

function(ouaexp_add_ui_test test_name source_file)
    ouaexp_register_test(${test_name} ${source_file} ${ARGN})
    target_sources(${test_name} PRIVATE $<TARGET_OBJECTS:ouaexp_test_resources>)
    target_link_libraries(${test_name} PRIVATE ouaexp_ui)
endfunction()

# PkiManager resolves its tree below QStandardPaths::GenericDataLocation, which
# every test process shares. These suites generate client certificates and move
# them between the trusted and rejected categories, so they must not overlap.
# APPEND matters here: ouaexp_tests_auth_integration already holds a lock of its
# own and a plain set_tests_properties() would drop it.
set_property(TEST
    ouaexp_tests
    ouaexp_tests_application
    ouaexp_tests_auth_integration
    ouaexp_tests_certificate_details_widget
    ouaexp_tests_certificates_dialog
    ouaexp_tests_connection_dialog
    ouaexp_tests_endpoint_settings_dialog
    ouaexp_tests_pkimanager
    APPEND PROPERTY RESOURCE_LOCK application_pki)

include(ProcessorCount)
ProcessorCount(OUAEXP_TEST_PARALLEL_LEVEL)
if(OUAEXP_TEST_PARALLEL_LEVEL EQUAL 0)
    set(OUAEXP_TEST_PARALLEL_LEVEL 1)
endif()

# Every test process isolates its own settings in a temporary directory and runs
# offscreen, so they are safe to run concurrently. The exceptions are the suites
# above, which share fixed server ports or the application PKI tree and are held
# apart by their RESOURCE_LOCKs.
get_property(test_targets GLOBAL PROPERTY OUAEXP_TEST_TARGETS)
add_custom_target(ouaexp_check
    COMMAND ${CMAKE_CTEST_COMMAND} --output-on-failure
            --parallel ${OUAEXP_TEST_PARALLEL_LEVEL} -L ouaexp
    DEPENDS ${test_targets}
    USES_TERMINAL
)
